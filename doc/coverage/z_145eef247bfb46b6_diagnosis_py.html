1"""Abweichungen als diagnostisches Signal (Abschnitt „Abweichungen als Chance"). 

2 

3Bayes-Indikator, Lern-Loop-Update und VOI-Bedingung. Alle Rechnungen sind exakt 

4(:class:`fractions.Fraction`). Undefinierte Fälle (z. B. Likelihood 0 unter ``H_0``) lösen 

5``ValueError`` aus; sie werden nicht stillschweigend zu einer Zahl.

20def posterior_odds(prior_h1: Rational, likelihood_h1: Rational, likelihood_h0: Rational) -> Fraction: 

21 """Posterior Odds ``P(H1|d)/P(H0|d) = P(H1)/P(H0) · p(d|H1)/p(d|H0)``.""" 

22 prior, l1, l0 = to_fraction(prior_h1), to_fraction(likelihood_h1), to_fraction(likelihood_h0) 

23 if not 0 < prior < 1: 

24 raise ValueError("Prior P(H1) muss in (0, 1) liegen") 

25 if l1 < 0 or l0 < 0: 

26 raise ValueError("Likelihoods müssen nichtnegativ sein") 

27 if l0 == 0: 

28 raise ValueError("p(d|H0) = 0: die Odds sind undefiniert") 

29 return prior / (1 - prior) * l1 / l0

32def posterior_probability(prior_h1: Rational, likelihood_h1: Rational, 

33 likelihood_h0: Rational) -> Fraction: 

34 """Posterior ``P(H1|d)`` aus den Odds.""" 

35 odds = posterior_odds(prior_h1, likelihood_h1, likelihood_h0) 

36 return odds / (1 + odds)

39def bayes_update(prior: Mapping[str, Rational], likelihood: Mapping[str, Rational]) -> Dict[str, Fraction]: 

40 """Lern-Loop: ``π₁(θ) ∝ p(d|θ)·π₀(θ)`` über einem endlichen Zustandsraum.""" 

41 if set(prior) != set(likelihood): 

42 raise ValueError("Prior und Likelihood müssen dieselben Zustände haben") 

43 pi0 = {state: to_fraction(p) for state, p in prior.items()} 

44 if any(p < 0 for p in pi0.values()) or sum(pi0.values()) != 1: 

45 raise ValueError("Prior muss nichtnegativ sein und sich zu 1 summieren") 

46 lik = {state: to_fraction(p) for state, p in likelihood.items()} 

47 if any(p < 0 for p in lik.values()): 

48 raise ValueError("Likelihoods müssen nichtnegativ sein") 

49 evidence = sum(lik[s] * pi0[s] for s in pi0) 

50 if evidence == 0: 

51 raise ValueError("Die beobachtete Abweichung hat Wahrscheinlichkeit 0 unter dem Prior") 

52 return {s: lik[s] * pi0[s] / evidence for s in pi0}

55def expected_gain(posterior: Mapping[str, Rational], gains: Mapping[str, Rational]) -> Fraction: 

56 """Erwarteter Zusatznutzen ``E_{π₁}[ΔW(θ)] = Σ_θ π₁(θ)·ΔW(θ)`` (Beweisskizze VOI).""" 

57 if set(posterior) != set(gains): 

58 raise ValueError("Posterior und Nutzen müssen dieselben Zustände haben") 

59 return sum((to_fraction(posterior[s]) * to_fraction(gains[s]) for s in posterior), Fraction(0))

62def integration_rational(expected_gain_value: Rational, cost: Rational) -> bool: 

63 """VOI-Bedingung: Erweiterung ist rational genau dann, wenn ``E[ΔW] ≥ C_int``.""" 

64 integration_cost = to_fraction(cost) 

65 if integration_cost < 0: 

66 raise ValueError("Integrationskosten müssen nichtnegativ sein") 

67 return to_fraction(expected_gain_value) >= integration_cost
